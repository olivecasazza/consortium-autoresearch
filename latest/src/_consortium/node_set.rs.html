1//! PyO3 wrappers for consortium::node_set.

3use pyo3::exceptions::{PyException, PyValueError};
4use pyo3::prelude::*;
5
6/// Python-visible NodeSet class.
7#[pyclass(name = "NodeSet")]
8#[derive(Debug, Clone)]
9pub struct PyNodeSet {
10    inner: consortium::node_set::NodeSet,
11}
12
13pyo3::create_exception!(ClusterShell.NodeSet, NodeSetException, PyException);
14pyo3::create_exception!(ClusterShell.NodeSet, NodeSetError, NodeSetException);
15pyo3::create_exception!(ClusterShell.NodeSet, NodeSetExternalError, NodeSetError);
16// NodeSetParseError stays on PyValueError: it predates this hierarchy, and
17// re-basing it would break any caller doing `except ValueError`. Upstream
18// derives it from NodeSetError instead; that divergence is left for the
19// NodeSet port to settle, not smuggled in here.
20pyo3::create_exception!(ClusterShell.NodeSet, NodeSetParseError, PyValueError);
21
22#[pymethods]
23impl PyNodeSet {
24    #[new]
25    #[pyo3(signature = (pattern=None))]
26    fn new(pattern: Option<&str>) -> PyResult<Self> {
27        match pattern {
28            Some(p) => {
29                let inner = consortium::node_set::NodeSet::parse(p)
30                    .map_err(|e| NodeSetParseError::new_err(e.to_string()))?;
31                Ok(Self { inner })
32            }
33            None => Ok(Self {
34                inner: consortium::node_set::NodeSet::new(),
35            }),
36        }
37    }
38
39    fn __str__(&self) -> String {
40        self.inner.to_string()
41    }
42
43    fn __repr__(&self) -> String {
44        format!("NodeSet(\"{}\")", self.inner)
45    }
46
47    fn __len__(&self) -> usize {
48        self.inner.len()
49    }
50
51    fn __contains__(&self, node: &str) -> bool {
52        self.inner.contains(node)
53    }
54
55    fn __or__(&self, other: &PyNodeSet) -> Self {
56        Self {
57            inner: self.inner.union(&other.inner),
58        }
59    }
60
61    fn __and__(&self, other: &PyNodeSet) -> Self {
62        Self {
63            inner: self.inner.intersection(&other.inner),
64        }
65    }
66
67    fn __sub__(&self, other: &PyNodeSet) -> Self {
68        Self {
69            inner: self.inner.difference(&other.inner),
70        }
71    }
72
73    fn __xor__(&self, other: &PyNodeSet) -> Self {
74        Self {
75            inner: self.inner.symmetric_difference(&other.inner),
76        }
77    }
78}

80/// Register node_set types into the parent module.
81pub fn register(parent: &Bound<'_, PyModule>) -> PyResult<()> {
82    parent.add_class::<PyNodeSet>()?;
83    // The exception hierarchy was declared but never added to the module, so
84    // `from ClusterShell.NodeSet import NodeSetExternalError` raised
85    // ImportError. ClusterShell.CLI.Error imports exactly that name, which is
86    // why all five CLI test modules died at collection under the Rust backend.
87    // NodeSetParseError was also unreachable, so the NodeSet shim's
88    // `from ClusterShell._consortium import NodeSet, NodeSetParseError` probe
89    // failed and silently degraded to the pure-Python NodeSet.
90    for (name, ty) in [
91        (
92            "NodeSetException",
93            parent.py().get_type_bound::<NodeSetException>(),
94        ),
95        ("NodeSetError", parent.py().get_type_bound::<NodeSetError>()),
96        (
97            "NodeSetExternalError",
98            parent.py().get_type_bound::<NodeSetExternalError>(),
99        ),
100        (
101            "NodeSetParseError",
102            parent.py().get_type_bound::<NodeSetParseError>(),
103        ),
104    ] {
105        parent.add(name, ty)?;
106    }
107    Ok(())
108}
