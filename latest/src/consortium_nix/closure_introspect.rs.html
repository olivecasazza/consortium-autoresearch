1//! Closure introspection — query the local nix store for what paths a
2//! given toplevel depends on.
3//!
4//! Used by [`crate::deploy_with_cascade`] to compute the *shared
5//! closure* across a heterogeneous fleet: the set of store paths that
6//! every host in the deploy will need regardless of its individual
7//! toplevel. Cascading those shared paths peer-to-peer (instead of one
8//! representative toplevel) gets the fan-out win without sending any
9//! host-specific bits to hosts that don't need them.
10//!
11//! ## What `nix path-info -r` returns
12//!
13//! `nix path-info -r /nix/store/AAA-system` walks AAA's reference graph
14//! and prints every store path AAA transitively depends on, one per
15//! line. This is the same closure `nix copy` would walk — so paths
16//! present in *every* toplevel's `path-info -r` output are guaranteed
17//! to be on every host post-cascade, and the per-host tail copy will
18//! only transfer the per-host-unique paths.
19//!
20//! ## Performance
21//!
22//! `nix path-info -r` walks the local store metadata only — no network
23//! IO, no hashing, just sqlite lookups. ~100–500ms per toplevel for
24//! typical NixOS systems. We run them in parallel across toplevels.
25
26use std::collections::HashSet;
27use std::process::Command;
28use std::sync::Mutex;
29use std::thread;
30
31use crate::error::{NixError, Result};
32
33/// Return the full closure of `store_path` — every store path it
34/// transitively depends on, including itself.
35///
36/// Runs `nix path-info -r STORE_PATH` against the local store. Errors
37/// if the path isn't present locally or if nix returns non-zero.
38pub fn closure_of(store_path: &str) -> Result<HashSet<String>> {
39    let output = Command::new("nix")
40        .args(["path-info", "-r", store_path])
41        .output()
42        .map_err(|e| NixError::General(format!("failed to spawn nix path-info: {e}")))?;
43
44    if !output.status.success() {
45        let stderr = String::from_utf8_lossy(&output.stderr);
46        return Err(NixError::General(format!(
47            "nix path-info -r {store_path} failed: {stderr}"
48        )));
49    }
50
51    let stdout = String::from_utf8_lossy(&output.stdout);
52    let paths: HashSet<String> = stdout
53        .lines()
54        .map(|l| l.trim())
55        .filter(|l| !l.is_empty())
56        .map(|l| l.to_string())
57        .collect();
58
59    Ok(paths)
60}
61
62/// Compute the set intersection of every toplevel's closure — paths
63/// present in ALL of them. These are the "shared substrate" paths
64/// every host in the deploy needs regardless of its individual
65/// toplevel; cascading them gets the fan-out win without sending any
66/// host-specific bits.
67///
68/// Runs `closure_of` for each toplevel in parallel. Returns an empty
69/// vec for empty input. If closure introspection fails for any
70/// toplevel the error propagates — caller decides whether to fall
71/// back to the carrier-toplevel approach.
72pub fn shared_paths(toplevels: &[String]) -> Result<Vec<String>> {
73    if toplevels.is_empty() {
74        return Ok(Vec::new());
75    }
76    if toplevels.len() == 1 {
77        // Single toplevel: its closure IS the shared closure.
78        let mut paths: Vec<String> = closure_of(&toplevels[0])?.into_iter().collect();
79        paths.sort();
80        return Ok(paths);
81    }
82
83    // Parallel introspection — each `nix path-info -r` is independent.
84    let results: Mutex<Vec<Result<HashSet<String>>>> =
85        Mutex::new(Vec::with_capacity(toplevels.len()));
86    thread::scope(|scope| {
87        let mut handles = Vec::with_capacity(toplevels.len());
88        for tl in toplevels {
89            let tl = tl.clone();
90            handles.push(scope.spawn(move || closure_of(&tl)));
91        }
92        for h in handles {
93            let r = h
94                .join()
95                .map_err(|_| NixError::General("closure_of thread panicked".into()))
96                .and_then(|inner| inner);
97            results.lock().unwrap().push(r);
98        }
99    });
100
101    let mut closures: Vec<HashSet<String>> = Vec::with_capacity(toplevels.len());
102    for r in results.into_inner().unwrap() {
103        closures.push(r?);
104    }
105
106    // Intersect — start with the smallest closure to keep the running
107    // intersection small.
108    closures.sort_by_key(|c| c.len());
109    let mut iter = closures.into_iter();
110    let mut acc = iter.next().unwrap();
111    for c in iter {
112        acc.retain(|p| c.contains(p));
113    }
114
115    let mut paths: Vec<String> = acc.into_iter().collect();
116    paths.sort();
117    Ok(paths)
118}
